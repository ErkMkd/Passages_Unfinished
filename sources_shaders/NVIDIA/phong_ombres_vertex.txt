
/************************************************************************************************************

		Affichage des ombres portées, avec les sources lumineuses d'openGl, sans texture
		Calcul les coordonnées UV de la texture de profondeur sur les sommets vus de la source génératrice des ombres
		
						!!! La source des ombres est obligatoirement la source 0 !!!

**************************************************************************************************************/


//------- Données de gestion des ombres:
uniform mat4 matrice_projection_vision_source;
uniform mat4 matrice_vision_source;
uniform mat4 matrice_source_uv;	//transformation_uv*projection*vision*matrice_volume

out vec4 coordonnees_texture_ombres;
out float distance_source_des_ombres;

//------ Données de gestion de la lumière:

uniform int drapeaux_sources;	//Les bits d'activation des sources lumineuses

out vec3 direction_oeil;
out vec3 directions_sources[8];
out float distances_sources[8];
out vec3 normale_face;

out vec3 couleur_ambiante_globale;

//====================================
//			Main
//====================================

void main( void )
{	
	//-------------- Calculs ombre:
		coordonnees_texture_ombres=(matrice_source_uv*sommet_position);
	
	//--------------- Position du sommet:
		gl_Position 	=  matrice_modele_vue_projection * sommet_position;
		normale_face	= matrice_normale * sommet_normale;
		vec3 position_sommet 	= (matrice_modele_vue * sommet_position).xyz;
		direction_oeil 	= -position_sommet;
   
   // ----------------- Calcul des sources:	
	distance_source_des_ombres=-(matrice_vision_source * sommet_position).z;
			
	for(int n=0;n<NBR_SOURCES;n++)
	{
		if ( (drapeaux_sources & (1<<n))==0 ) continue;
		vec3 position_source 		= (sources[n].position).xyz;
		directions_sources[n] 		= position_source-position_sommet;
		distances_sources[n] 		= length(directions_sources[n]);
	}
		
	couleur_ambiante_globale = materiau_ambiante.rgb*ambiante_globale.rgb;	
}
