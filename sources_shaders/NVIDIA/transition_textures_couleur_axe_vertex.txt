//======================================================================
//				Transition entre deux textures de couleur,
//				le long de l'axe Z du volume
//				Vertex shader
//======================================================================

//Données d'éclairage:
uniform int drapeaux_sources;	//Les bits d'activation des sources lumineuses

out vec4 couleur;
out vec3 couleur_speculaire; //Spérarée pour apparaître par dessus les textures

// Données de transition:
uniform float zmin;
uniform float zmax;

uniform int drapeaux_textures;

out vec2 coordonnees_texture_depart;
out vec2 coordonnees_texture_destination;
out float position_axe; //Varie de 0 à 1

void main( void )
{
	//----------Calculs pour la transition de textures:
    gl_Position =  matrice_modele_vue_projection * sommet_position;
	position_axe=(sommet_position.z-zmin)/(zmax-zmin);
	
	if((drapeaux_textures&0x1)!=0) coordonnees_texture_depart     	= sommet_texture0.xy;
	if((drapeaux_textures&0x2)!=0) coordonnees_texture_destination   = sommet_texture1.xy;
	
	//------------ Eclairage:
	
			//Données de l'observateur:
			vec3 position_sommet = (matrice_modele_vue * sommet_position).xyz;
			vec3 normale 				= matrice_normale * sommet_normale;
			vec3 direction_oeil 	= -normalize(position_sommet);
			
			//Sources lumineuses:
			couleur_speculaire=vec3 (0.,0.,0.);
			vec3 couleur_rgb=vec3(0.,0.,0.);
			vec3 couleur_ambiante_globale = materiau_ambiante.rgb*ambiante_globale.rgb;
			
			for(int n=0;n<NBR_SOURCES;n++)
			{
				if ( (drapeaux_sources & (1<<n))==0 ) continue;
				
				vec3 position_source 		= (sources[n].position).xyz;
				vec3 direction_source 		= position_source-position_sommet;
				float distance_source 		= length(direction_source);
				direction_source			= normalize(direction_source);
				float angle_source_normale  = max(dot(normale,direction_source),0.);
				vec3 reflet_source			= normalize(direction_oeil+direction_source);
				
				
				//Composantes:
			
				float attenuation		=	1./ (sources[n].attenuation_constante 
											+ sources[n].attenuation_lineaire * distance_source
											+  distance_source * distance_source * sources[n].attenuation_quadratique);
			
				vec3 couleur_ambiante   = 	materiau_ambiante.rgb*sources[n].ambiante.rgb;
		   
				vec3 couleur_diffuse 	= 	sources[n].diffuse.rgb * materiau_diffuse.rgb * angle_source_normale;
				
				couleur_rgb				+=	attenuation * (couleur_ambiante + couleur_diffuse);
			
				if (angle_source_normale>0.)
				{
					couleur_speculaire 	+= attenuation* (
															sources[n].speculaire.rgb * materiau_speculaire.rgb
															* pow(max(dot(reflet_source,normale),0.),materiau_brillance)
														);
				}
				
			}
		
		//----------- Couleur de base:
		couleur=vec4(materiau_emission.rgb + couleur_ambiante_globale + couleur_rgb , materiau_diffuse.a);	
}
