uniform int taille_flou;
uniform float saturation_flou;
uniform sampler2D texture_source;

in vec2 coordonnees_texture;

void main(void)
{
   int dy;
   float taille_texture=textureSize(texture_source,0).y;
   float intensite;
   float amplitude=0.;
   vec4 p=vec4 ( 0.);
   for (dy=-taille_flou;dy<taille_flou;dy++)
   {
      intensite=1.-abs(float(dy)/float(taille_flou));
      p+= intensite *(texture(texture_source,vec2(coordonnees_texture.x,coordonnees_texture.y+float(dy)/taille_texture)));
      amplitude+=intensite;
   }
   p*=saturation_flou/amplitude;
   p.a=1.;
   
   couleur_fragment = p;
}
